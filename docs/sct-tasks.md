# Supply Chain Tracker — Blockchain Workflow & Task List

> **BUILD STATUS: the project is built. Phases 1–9 are done in the repository (one git commit per phase).**
> The sections below are the original design. The working code lives in the repo; where they differ, **the repo wins**.
>
> | Phase | Status | Notes |
> |---|---|---|
> | 1–3 Setup, contracts, tests | Done | 13 contract tests pass |
> | 4 Backend | Done | Wallet nonce login, role middleware, event poller; API e2e test passes |
> | 5 Frontend | Done | Written and served; **needs your manual MetaMask test** (docs/TESTING.md) |
> | 6 E2E testing | Partly manual | Automated parts done; browser checklist is yours |
> | 7 Sepolia | Prepared | Needs your Infura key, wallet and faucet ETH (docs/DEPLOY.md) |
> | 8 Hardening | Done in code + notes | Contracts unaudited; see SECURITY.md |
> | 9 Docs | Done | README, guides, license, CI |
>
> **Differences from the design below**
> - No Truffle (it was sunset in 2023). `solc-js` compiles, plain Web3 scripts deploy: `npm run compile`, `npm run deploy:dev`, `npm run deploy:sepolia`. Build output is in `build/`, not `artifacts/`.
> - Tests use Node's built-in test runner with an in-process Ganache, not Truffle's `contract()` helper.
> - Auth is a one-time nonce + signature + session token (Phase 8 hardening is already included), not the simple header check shown in Section 7.
> - Events are polled (Web3 1.x cannot subscribe over HTTP).
> - The frontend loads Web3 from the server (`/vendor/web3.min.js`) because of the strict Content-Security-Policy.

## 4. Smart Contracts

### 4.1 `contracts/RoleManager.sol`

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract RoleManager {
    address public admin;

    enum Role { None, Manufacturer, Distributor, Retailer, Inspector }

    mapping(address => Role) public roles;

    event RoleAssigned(address indexed account, Role role);
    event RoleRevoked(address indexed account);

    modifier onlyAdmin() {
        require(msg.sender == admin, "Not admin");
        _;
    }

    constructor() {
        admin = msg.sender;
        roles[msg.sender] = Role.Inspector; // admin is also inspector
    }

    function assignRole(address account, Role role) external onlyAdmin {
        roles[account] = role;
        emit RoleAssigned(account, role);
    }

    function revokeRole(address account) external onlyAdmin {
        roles[account] = Role.None;
        emit RoleRevoked(account);
    }

    function getRole(address account) external view returns (Role) {
        return roles[account];
    }

    function hasRole(address account, Role role) external view returns (bool) {
        return roles[account] == role;
    }
}
```

### 4.2 `contracts/SupplyChain.sol`

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "./RoleManager.sol";

contract SupplyChain {
    RoleManager public roleManager;

    // ── Enums ──────────────────────────────────────────────
    enum ProductStatus {
        Created,        // 0
        InTransit,      // 1
        AtDistributor,  // 2
        AtRetailer,     // 3
        Sold,           // 4
        Recalled        // 5
    }

    // ── Structs ────────────────────────────────────────────
    struct Product {
        uint256 id;
        string name;
        string batchNumber;
        string origin;          // factory / farm / source
        address manufacturer;
        uint256 manufacturedAt; // unix timestamp
        uint256 expiryDate;     // unix timestamp (0 if N/A)
        ProductStatus status;
        bool exists;
    }

    struct Checkpoint {
        address actor;
        string location;
        string notes;
        ProductStatus status;
        uint256 timestamp;
    }

    // ── State ──────────────────────────────────────────────
    uint256 private _productCounter;

    mapping(uint256 => Product) public products;
    mapping(uint256 => Checkpoint[]) public checkpoints;

    // ── Events ─────────────────────────────────────────────
    event ProductCreated(
        uint256 indexed productId,
        string name,
        string batchNumber,
        address indexed manufacturer
    );

    event CheckpointAdded(
        uint256 indexed productId,
        address indexed actor,
        string location,
        ProductStatus status,
        uint256 timestamp
    );

    event ProductRecalled(uint256 indexed productId, address indexed by);

    // ── Modifiers ──────────────────────────────────────────
    modifier productExists(uint256 productId) {
        require(products[productId].exists, "Product not found");
        _;
    }

    modifier onlyRole(RoleManager.Role role) {
        require(
            roleManager.hasRole(msg.sender, role),
            "Unauthorized role"
        );
        _;
    }

    // ── Constructor ────────────────────────────────────────
    constructor(address roleManagerAddress) {
        roleManager = RoleManager(roleManagerAddress);
    }

    // ── Functions ──────────────────────────────────────────

    /// @notice Manufacturer registers a new product
    function createProduct(
        string calldata name,
        string calldata batchNumber,
        string calldata origin,
        uint256 expiryDate
    ) external onlyRole(RoleManager.Role.Manufacturer) returns (uint256) {
        _productCounter++;
        uint256 id = _productCounter;

        products[id] = Product({
            id: id,
            name: name,
            batchNumber: batchNumber,
            origin: origin,
            manufacturer: msg.sender,
            manufacturedAt: block.timestamp,
            expiryDate: expiryDate,
            status: ProductStatus.Created,
            exists: true
        });

        checkpoints[id].push(Checkpoint({
            actor: msg.sender,
            location: origin,
            notes: "Product manufactured",
            status: ProductStatus.Created,
            timestamp: block.timestamp
        }));

        emit ProductCreated(id, name, batchNumber, msg.sender);
        return id;
    }

    /// @notice Distributor picks up the product — marks InTransit
    function shipProduct(
        uint256 productId,
        string calldata location,
        string calldata notes
    ) external onlyRole(RoleManager.Role.Distributor) productExists(productId) {
        Product storage p = products[productId];
        require(
            p.status == ProductStatus.Created || p.status == ProductStatus.AtDistributor,
            "Cannot ship from current status"
        );

        p.status = ProductStatus.InTransit;
        _addCheckpoint(productId, location, notes, ProductStatus.InTransit);
    }

    /// @notice Distributor receives product at warehouse
    function receiveAtDistributor(
        uint256 productId,
        string calldata location,
        string calldata notes
    ) external onlyRole(RoleManager.Role.Distributor) productExists(productId) {
        Product storage p = products[productId];
        require(p.status == ProductStatus.InTransit, "Must be InTransit");

        p.status = ProductStatus.AtDistributor;
        _addCheckpoint(productId, location, notes, ProductStatus.AtDistributor);
    }

    /// @notice Retailer receives and stocks product
    function receiveAtRetailer(
        uint256 productId,
        string calldata location,
        string calldata notes
    ) external onlyRole(RoleManager.Role.Retailer) productExists(productId) {
        Product storage p = products[productId];
        require(
            p.status == ProductStatus.InTransit || p.status == ProductStatus.AtDistributor,
            "Invalid prior status"
        );

        p.status = ProductStatus.AtRetailer;
        _addCheckpoint(productId, location, notes, ProductStatus.AtRetailer);
    }

    /// @notice Retailer marks product as sold
    function markSold(
        uint256 productId,
        string calldata notes
    ) external onlyRole(RoleManager.Role.Retailer) productExists(productId) {
        Product storage p = products[productId];
        require(p.status == ProductStatus.AtRetailer, "Must be at retailer");

        p.status = ProductStatus.Sold;
        _addCheckpoint(productId, "POS", notes, ProductStatus.Sold);
    }

    /// @notice Inspector can recall a product at any stage
    function recallProduct(
        uint256 productId,
        string calldata reason
    ) external onlyRole(RoleManager.Role.Inspector) productExists(productId) {
        products[productId].status = ProductStatus.Recalled;
        _addCheckpoint(productId, "N/A", reason, ProductStatus.Recalled);
        emit ProductRecalled(productId, msg.sender);
    }

    // ── Read functions ──────────────────────────────────────

    function getProduct(uint256 productId)
        external
        view
        productExists(productId)
        returns (Product memory)
    {
        return products[productId];
    }

    function getCheckpoints(uint256 productId)
        external
        view
        productExists(productId)
        returns (Checkpoint[] memory)
    {
        return checkpoints[productId];
    }

    function getCheckpointCount(uint256 productId)
        external
        view
        returns (uint256)
    {
        return checkpoints[productId].length;
    }

    function getTotalProducts() external view returns (uint256) {
        return _productCounter;
    }

    // ── Internal ────────────────────────────────────────────

    function _addCheckpoint(
        uint256 productId,
        string memory location,
        string memory notes,
        ProductStatus status
    ) internal {
        checkpoints[productId].push(Checkpoint({
            actor: msg.sender,
            location: location,
            notes: notes,
            status: status,
            timestamp: block.timestamp
        }));

        emit CheckpointAdded(productId, msg.sender, location, status, block.timestamp);
    }
}
```

### `backend/utils/signatureVerify.js` — MetaMask Auth

```js
const web3 = require('../config/web3');

/**
 * Verifies that `signature` was produced by `address`
 * signing `message`. Used to authenticate API requests.
 */
function verifySignature(message, signature, address) {
  const recovered = web3.eth.accounts.recover(message, signature);
  return recovered.toLowerCase() === address.toLowerCase();
}

module.exports = { verifySignature };
```

### `backend/middleware/auth.middleware.js`

```js
const { verifySignature } = require('../utils/signatureVerify');

module.exports = function (req, res, next) {
  const { address, signature, message } = req.headers;
  if (!address || !signature || !message) {
    return res.status(401).json({ error: 'Missing auth headers' });
  }
  if (!verifySignature(message, signature, address)) {
    return res.status(403).json({ error: 'Invalid signature' });
  }
  req.signerAddress = address;
  next();
};
```

```js
async function trackProduct(productId) {
  const [productRes, checkpointRes] = await Promise.all([
    fetch(`/api/products/${productId}`),
    fetch(`/api/products/${productId}/checkpoints`),
  ]);

  const product     = await productRes.json();
  const checkpoints = await checkpointRes.json();

  // Render product info
  document.getElementById('productName').textContent = product.name;
  document.getElementById('productStatus').textContent = statusLabel(product.status);

  // Render checkpoint timeline
  const timeline = document.getElementById('timeline');
  timeline.innerHTML = '';
  checkpoints.forEach(cp => {
    const li = document.createElement('li');
    li.textContent = `[${new Date(cp.timestamp * 1000).toLocaleString()}] 
      ${cp.location} — ${cp.notes} (by ${cp.actor})`;
    timeline.appendChild(li);
```

### `frontend/public/js/updateShipment.js`

```js
async function updateShipment(action, productId, location, notes) {
  const { web3, userAccount } = await initWeb3();

  const res  = await fetch('/api/abi/SupplyChain');
  const { abi, address } = await res.json();
  const contract = new web3.eth.Contract(abi, address);

  const methodMap = {
    ship:               contract.methods.shipProduct(productId, location, notes),
    receiveDistributor: contract.methods.receiveAtDistributor(productId, location, notes),
    receiveRetailer:    contract.methods.receiveAtRetailer(productId, location, notes),
    markSold:           contract.methods.markSold(productId, notes),
    recall:             contract.methods.recallProduct(productId, notes),
  };

  const method = methodMap[action];
  if (!method) throw new Error('Unknown action');

  await method.send({ from: userAccount });
  alert(`Action '${action}' sent to blockchain`);
}
```
